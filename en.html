Preparing the graph
Drag nodes · Hover to highlight
relationship.ts
import { Graph } from '@antv/g6-lite';

// HTML: <div id="container" style="height: 360px"></div>
const data = {
  nodes: [
    {"id":"G6Lite","x":280,"y":200,"data":{"fill":"#596bc0","stroke":"#e2e8ff","labelColor":"#43558b"}},
    {"id":"React","x":135,"y":80,"data":{"fill":"#7e9fd5","stroke":"#e7efff","labelColor":"#315f9c"}},
    {"id":"Vue","x":280,"y":45,"data":{"fill":"#6db49d","stroke":"#e1f4eb","labelColor":"#246d58"}},
    {"id":"TypeScript","x":425,"y":80,"data":{"fill":"#7e9fd5","stroke":"#e7efff","labelColor":"#315f9c"}},
    {"id":"WebGL","x":465,"y":235,"data":{"fill":"#d6ab60","stroke":"#fff0d8","labelColor":"#87601f"}},
    {"id":"Layouts","x":365,"y":350,"data":{"fill":"#d38ba8","stroke":"#fbe6ef","labelColor":"#98496b"}},
    {"id":"Data","x":195,"y":350,"data":{"fill":"#6db49d","stroke":"#e1f4eb","labelColor":"#246d58"}},
    {"id":"Interaction","x":95,"y":235,"data":{"fill":"#d38ba8","stroke":"#fbe6ef","labelColor":"#98496b"}}
  ],
  edges: [
    {"source":"G6Lite","target":"React"},
    {"source":"G6Lite","target":"Vue"},
    {"source":"G6Lite","target":"TypeScript"},
    {"source":"G6Lite","target":"WebGL"},
    {"source":"G6Lite","target":"Layouts"},
    {"source":"G6Lite","target":"Data"},
    {"source":"G6Lite","target":"Interaction"}
  ],
};

const graph = new Graph({
  container: '#container',
  data,
  "theme": {
    "type": "light",
    "canvasBackground": "#ffffff"
  },
  "autoFit": {
    "padding": 24
  },
  "label": {
    "scaleWithZoom": false
  },
  "node": {
    "type": "circle",
    "style": {
      "size": 28,
      "fill": a=>String(a.data?.fill??"#7087cf"),
      "stroke": a=>String(a.data?.stroke??"#c7d1f0"),
      "strokeWidth": 6,
      "label": a=>a.id,
      "labelPlacement": "bottom",
      "labelFontSize": 13,
      "labelColor": a=>String(a.data?.labelColor??"#3d508e")
    }
  },
  "edge": {
    "type": "line",
    "style": {
      "stroke": (a,b)=>String(b.data?.fill??"#9baed4"),
      "lineWidth": 1.5
    }
  },
  "behaviors": [
    "drag-canvas",
    "drag-node",
    "hover-highlight"
  ]
});

await graph.render();

// Call graph.destroy() when the component unmounts.

Layers that work as one

Graph manages data, layouts, and interactions. WebGL renders nodes and edges, while Canvas 2D draws labels, icons, and group boundaries. A shared viewport and scheduler keep the layers aligned.
