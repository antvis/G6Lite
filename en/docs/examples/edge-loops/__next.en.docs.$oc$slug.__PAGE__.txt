        fill: '#5B8FF9',
        size: 36,
        label: (node) => node.name ?? node.id,
        labelPlacement: 'bottom',
      },
    },
    edge: {
      type: 'line',
      style: { stroke: '#99ADD1', lineWidth: 2 },

  });
