import { Graph } from '@antv/g6-lite';

// HTML: <div id="container" style="height: 360px"></div>
const data = {
  nodes: [
    {"id":"G6Lite","x":280,"y":200,"data":{"fill":"#596bc0","stroke":"#e2e8ff","labelColor":"#43558b"}},
    {"id":"React","x":135,"y":80,"data":{"fill":"#7e9fd5","stroke":"#e7efff","labelColor":"#315f9c"}},
    {"id":"Vue","x":280,"y":45,"data":{"fill":"#6db49d","stroke":"#e1f4eb","labelColor":"#246d58"}},
    {"id":"TypeScript","x":425,"y":80,"data":{"fill":"#7e9fd5","stroke":"#e7efff","labelColor":"#315f9c"}},
    {"id":"WebGL","x":465,"y":235,"data":{"fill":"#d6ab60","stroke":"#fff0d8","labelColor":"#87601f"}},
    {"id":"布局","x":365,"y":350,"data":{"fill":"#d38ba8","stroke":"#fbe6ef","labelColor":"#98496b"}},
    {"id":"数据","x":195,"y":350,"data":{"fill":"#6db49d","stroke":"#e1f4eb","labelColor":"#246d58"}},
    {"id":"交互","x":95,"y":235,"data":{"fill":"#d38ba8","stroke":"#fbe6ef","labelColor":"#98496b"}}
  ],
  edges: [
    {"source":"G6Lite","target":"React"},
    {"source":"G6Lite","target":"Vue"},
    {"source":"G6Lite","target":"TypeScript"},
    {"source":"G6Lite","target":"WebGL"},
    {"source":"G6Lite","target":"布局"},
    {"source":"G6Lite","target":"数据"},
    {"source":"G6Lite","target":"交互"}
  ],
};

const graph = new Graph({
  container: '#container',
  data,
  "theme": {
    "type": "light",
    "canvasBackground": "#ffffff"
  },
  "autoFit": {
    "padding": 24
  },
  "label": {
    "scaleWithZoom": false
  },
  "node": {
    "type": "circle",
    "style": {
      "size": 28,
      "fill": a=>String(a.data?.fill??"#7087cf"),
      "stroke": a=>String(a.data?.stroke??"#c7d1f0"),
      "strokeWidth": 6,
      "label": a=>a.id,
      "labelPlacement": "bottom",
      "labelFontSize": 13,
      "labelColor": a=>String(a.data?.labelColor??"#3d508e")
    }
  },
  "edge": {
    "type": "line",
    "style": {
      "stroke": (a,b)=>String(b.data?.fill??"#9baed4"),
      "lineWidth": 1.5
    },
    "stateStyles": {
      "highlight": {
        "stroke": "#4568c4",
        "lineWidth": 3,
        "opacity": 1
      }
    }
  },
  "behaviors": [
    "drag-canvas",
    "drag-node",
    "highlight-node"
  ]
});

await graph.render();

let highlightedEdges: string[] = [];
const clearHighlight = () => {
  graph.batchSetState(highlightedEdges.map((id) => ({ id, state: 'highlight', value: false })));
  highlightedEdges = [];
};
graph.on('node:pointerover', ({ nodeId }) => {
  clearHighlight();
  highlightedEdges = graph.getEdges()
    .filter((edge) => edge.source === nodeId || edge.target === nodeId)
    .map((edge) => edge.id);
  graph.batchSetState(highlightedEdges.map((id) => ({ id, state: 'highlight', value: true })));
});
graph.on('node:pointerleave', clearHighlight);


// 组件卸载时调用 graph.destroy()。

分层架构，协同渲染

Graph 协同组织数据管理、布局计算与交互控制。WebGL 批量绘制节点与边，Canvas 2D 承载标签、图标与分组边界，各图层共享视口与更新调度。
