The React and Vue 3 examples below create the same graph. Zoom the canvas, drag nodes, and hover to highlight connections.

<QuickStartExample locale="en" />

Run `npm install @antv/g6-lite` in your existing framework project, then use the matching component below.

## Use with React

Save this as `GraphView.tsx` and use `<GraphView />` in your page. The component creates the graph after mounting and destroys it on unmount.

```tsx
import { useEffect, useRef } from 'react';
import { Graph } from '@antv/g6-lite';

export function GraphView() {
  const containerRef = useRef<HTMLDivElement | null>(null);

  useEffect(() => {
    if (!containerRef.current) return;

    const graph = new Graph({
      container: containerRef.current,
      data: {
        nodes: [{ id: 'Alice' }, { id: 'Bob' }, { id: 'Carol' }],
        edges: [
          { source: 'Alice', target: 'Bob' },
          { source: 'Alice', target: 'Carol' },
        ],
      },
      node: {
        style: {
          size: 24,
          label: (node) => node.id,
          labelPlacement: 'bottom',
        },
      },
      edge: { style: { endArrow: true } },
      layout: { type: 'circular', radius: 120 },
      autoFit: { padding: 32, maxZoom: 1 },
      behaviors: ['zoom-canvas', 'drag-canvas', 'drag-node', 'hover-highlight'],
    });

    void graph.render().catch(console.error);

    return () => graph.destroy();
  }, []);

  return <div ref={containerRef} style={{ width: '100%', height: 360 }} />;
}
```

## Use with Vue

Save this as `GraphView.vue`, then import and use `<GraphView />` in your page. A template ref provides the container, and lifecycle hooks create and destroy the graph.

```vue
<script setup lang="ts">
import { onBeforeUnmount, onMounted, ref } from 'vue';
import { Graph } from '@antv/g6-lite';

const containerRef = ref<HTMLDivElement | null>(null);
let graph: Graph | undefined;

onMounted(() => {
  if (!containerRef.value) return;

  graph = new Graph({
    container: containerRef.value,
    data: {
      nodes: [{ id: 'Alice' }, { id: 'Bob' }, { id: 'Carol' }],
      edges: [
        { source: 'Alice', target: 'Bob' },
        { source: 'Alice', target: 'Carol' },
      ],
    },
    node: {
      style: {
        size: 24,
        label: (node) => node.id,
        labelPlacement: 'bottom',
      },
    },
    edge: { style: { endArrow: true } },
    layout: { type: 'circular', radius: 120 },
    autoFit: { padding: 32, maxZoom: 1 },
    behaviors: ['zoom-canvas', 'drag-canvas', 'drag-node', 'hover-highlight'],
  });

  void graph.render().catch(console.error);
});

onBeforeUnmount(() => {
  graph?.destroy();
});
</script>

<template>
  <div ref="containerRef" style="width: 100%; height: 360px" />
</template>
```
