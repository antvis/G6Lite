/**
 * Demo: 节点标签 / 边标签 / 节点图标
 */
import { Graph, type GraphEdge, type GraphNode } from '@antv/g6-lite';

const DATA = {
  nodes: [
    { id: 'server', data: { name: 'API Server', type: 'server' } },
    { id: 'db', data: { name: 'Database', type: 'database' } },
    { id: 'cache', data: { name: 'Redis Cache', type: 'cache' } },
    { id: 'queue', data: { name: 'Message Queue', type: 'queue' } },
    { id: 'worker', data: { name: 'Worker', type: 'worker' } },
    { id: 'gateway', data: { name: 'API Gateway', type: 'gateway' } },
    { id: 'client', data: { name: 'Client App', type: 'client' } },
    { id: 'monitor', data: { name: 'Monitor', type: 'monitor' } },
  ],
  edges: [
    { source: 'client', target: 'gateway', data: { label: 'HTTPS' } },
    { source: 'gateway', target: 'server', data: { label: 'gRPC' } },
    { source: 'server', target: 'db', data: { label: 'SQL' } },
    { source: 'server', target: 'cache', data: { label: 'TCP' } },
    { source: 'server', target: 'queue', data: { label: 'AMQP' } },
    { source: 'queue', target: 'worker', data: { label: 'consume' } },
    { source: 'worker', target: 'db', data: { label: 'write' } },
    { source: 'monitor', target: 'server', data: { label: 'metrics' } },
    { source: 'monitor', target: 'gateway', data: { label: 'health' } },
  ],
};

export async function run(container: string | HTMLElement) {
  const el = typeof container === 'string' ? document.querySelector<HTMLElement>(container)! : container;

  const toolbar = document.createElement('div');
  toolbar.style.cssText = 'position:absolute;top:8px;left:8px;z-index:10;display:flex;gap:6px;flex-wrap:wrap;';
  toolbar.innerHTML = `
    <button id="btn-label-bottom">标签: bottom</button>
    <button id="btn-label-center">标签: center</button>
    <button id="btn-label-top">标签: top</button>
    <button id="btn-label-right">标签: right</button>
    <button id="btn-toggle-edge-labels">切换边标签</button>
    <button id="btn-layout-circular">circular</button>
    <button id="btn-layout-force">force</button>
  `;
  el.style.position = 'relative';
  el.appendChild(toolbar);

  const graph = new Graph({
    autoFitOnResize: true,
    container: el,
    data: DATA,
    layout: { type: 'circular', radius: 180 },
    behaviors: ['zoom-canvas', 'drag-canvas', 'drag-node'],
    node: {
      style: {
        size: 20,
        fill: '#5B8FF9',
        label: (n) => (n.data?.name as string | undefined) ?? n.id,
        labelPlacement: 'bottom',
        labelFontSize: 12,
        labelColor: '#333',
      },
    },
    edge: {
      style: {
        label: (s, t, e) => (e.data?.label as string) ?? '',
        labelFontSize: 10,
        labelColor: '#666',
        labelAutoRotate: true,
      },
    },
    theme: 'light',
  });
  await graph.render();

  // Toolbar handlers
  document.getElementById('btn-label-bottom')!.onclick = () => {
    graph.updateOptions({ node: { style: { labelPlacement: 'bottom' } } });
  };
  document.getElementById('btn-label-center')!.onclick = () => {
    graph.updateOptions({ node: { style: { labelPlacement: 'center' } } });
  };
  document.getElementById('btn-label-top')!.onclick = () => {
    graph.updateOptions({ node: { style: { labelPlacement: 'top' } } });
  };
  document.getElementById('btn-label-right')!.onclick = () => {
    graph.updateOptions({ node: { style: { labelPlacement: 'right' } } });
  };

  let edgeLabelsVisible = true;
  document.getElementById('btn-toggle-edge-labels')!.onclick = () => {
    edgeLabelsVisible = !edgeLabelsVisible;
    graph.updateOptions({
      edge: {
        style: {
          label: edgeLabelsVisible
            ? ((_source: GraphNode, _target: GraphNode, edge: GraphEdge) =>
                (edge.data?.label as string | undefined) ?? '')
            : false,
        },
      },
    });
  };

  document.getElementById('btn-layout-circular')!.onclick = async () => {
    graph.setLayout({ type: 'circular', radius: 180 });
    await graph.layout();
    graph.fitView();
  };

  document.getElementById('btn-layout-force')!.onclick = async () => {
    graph.setLayout({ type: 'd3-force' });
    await graph.layout();
    graph.fitView();
  };

  return graph;
}
