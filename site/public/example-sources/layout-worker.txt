/**
 * Demo: run a force layout in a Web Worker.
 */
import { Graph } from '@antv/g6-lite';

function createNetwork(count: number) {
  const nodes = Array.from({ length: count }, (_, index) => ({
    id: `n${index}`,
    data: { group: index % 4 },
  }));
  const edges = nodes.slice(1).map((node, index) => ({
    source: `n${index % 8}`,
    target: node.id,
  }));
  return { nodes, edges };
}

export async function run(container: string | HTMLElement) {
  const host = typeof container === 'string' ? document.querySelector<HTMLElement>(container)! : container;
  host.style.position = 'relative';
  const started = performance.now();
  const graph = new Graph({
    autoFitOnResize: true,
    container: host,
    data: createNetwork(80),
    node: {
      style: {
        size: 10,
        fill: (node) => ['#5B8FF9', '#5AD8A6', '#F6BD16', '#E86452'][Number(node.data?.group ?? 0)],
      },
    },
    edge: { style: { stroke: '#D0D7E2', lineWidth: 1 } },
    layout: { type: 'force', worker: true, linkDistance: 48, maxIteration: 80 },
    behaviors: ['zoom-canvas', 'drag-canvas'],
  });
  await graph.render();

  const note = document.createElement('div');
  note.style.cssText =
    'position:absolute;left:12px;bottom:12px;z-index:20;color:#64748b;font:12px/1.4 system-ui,sans-serif;pointer-events:none;';
  note.textContent = `force · worker · ${Math.round(performance.now() - started)} ms`;
  host.append(note);

  return {
    fitView: () => graph.fitView(),
    destroy() {
      graph.destroy();
      note.remove();
    },
  };
}
